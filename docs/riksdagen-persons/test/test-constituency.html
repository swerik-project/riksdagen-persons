498                total=pl.len(),
499                filled=(pl.col("district_clean") != "").sum(),
500            )
501            .with_columns(
502                missing=pl.col("total") - pl.col("filled"),
503                completion_rate_value=pl.col("filled") / pl.col("total"),

508        all_years = []
509        incomplete_years = []

514                "total": int(row["total"]),
515                "filled": int(row["filled"]),
516                "missing": int(row["missing"]),
517                "completion_rate": f"{completion_rate:.2%}",
518            }
519            all_years.append(entry)

485    def get_constituency_completion_by_start_year(self, members):
486        """Calculate constituency completion by mandate start year."""
487        prepared_members = members.with_columns(
488            pl.col("start").cast(pl.Utf8).str.slice(0, 4).alias("year"),
489            pl.col("district")
490            .fill_null("")
491            .cast(pl.Utf8)
492            .str.strip_chars()
493            .alias("district_clean"),
494        )
495        completion_by_year = (
496            prepared_members
497            .group_by("year")
498            .agg(
499                total=pl.len(),
500                filled=(pl.col("district_clean") != "").sum(),
501            )
502            .with_columns(
503                missing=pl.col("total") - pl.col("filled"),
504                completion_rate_value=pl.col("filled") / pl.col("total"),
505            )
506            .sort("year")
507        )
508
509        all_years = []
510        incomplete_years = []
511        for row in completion_by_year.to_dicts():
512            completion_rate = round(float(row["completion_rate_value"]), 4)
513            entry = {
514                "year": str(row["year"]),
515                "total": int(row["total"]),
516                "filled": int(row["filled"]),
517                "missing": int(row["missing"]),
518                "completion_rate": f"{completion_rate:.2%}",
519            }
520            all_years.append(entry)
521            if completion_rate < 1.0:
522                incomplete_years.append(entry)
523        return all_years, incomplete_years
