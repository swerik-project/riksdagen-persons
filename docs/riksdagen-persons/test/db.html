 38def parse_date_interval(value, is_end):
 39    """
 40    Parse partial date strings as interval boundaries for integrity checks.
 41
 42    Start dates are expanded to the first possible day. End dates are expanded
 43    to the exclusive upper bound after the last possible day.
 44    """
 45    if pd.isna(value) or str(value).strip() == "":
 46        if is_end:
 47            return pd.Timestamp.max.normalize(), None, "blank"
 48        return None, None, "blank"
 49
 50    value = str(value).strip()
 51    match = DATE_RE.match(value)
 52    if match is None:
 53        return None, None, "malformed"
 54
 55    year = int(match.group(1))
 56    month = int(match.group(2)) if match.group(2) else None
 57    day = int(match.group(3)) if match.group(3) else None
 58
 59    try:
 60        if month is None:
 61            if is_end:
 62                return pd.Timestamp(year + 1, 1, 1), "year", None
 63            return pd.Timestamp(year, 1, 1), "year", None
 64
 65        if day is None:
 66            if is_end:
 67                if month == 12:
 68                    return pd.Timestamp(year + 1, 1, 1), "month", None
 69                return pd.Timestamp(year, month + 1, 1), "month", None
 70            return pd.Timestamp(year, month, 1), "month", None
 71
 72        return pd.Timestamp(year, month, day), "day", None
 73    except ValueError:
 74        return None, None, "malformed"

179        """
180        if pd.isna(value) or value == "":
181            return None
182
183        value = str(value)
184        for date_format in ("%Y-%m-%d", "%Y"):
185            try:
